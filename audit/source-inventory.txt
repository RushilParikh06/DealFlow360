          textarea.value = textarea.value ? (textarea.value.trim() + ' | ' + tag) : tag;

        this.classList.toggle('bg-primary');
        this.classList.toggle('text-on-primary');

        if (!textarea.value.trim()) {

          alert('Governance Policy: Review notes are required before routing to Finance.');
          return;
        }
        this.disabled = true;
        this.innerHTML = '<span class="material-symbols-outlined text-[18px]">sync</span> Routing...';
        setTimeout(() => {
          this.innerHTML = '<span class="material-symbols-outlined text-[18px]">check_circle</span> Routed to Stage 3';
          this.classList.remove('bg-primary');
          this.classList.add('bg-surface-container-high', 'text-on-surface');
        }, 600);
      });
    }

    if (rejectBtn) {
      rejectBtn.addEventListener('click', function() {
        if (!textarea.value.trim()) {
          textarea.focus();
          textarea.classList.add('border-error');
          alert('Governance Policy: State specific reason for rejection.');
          return;
        }
        confirm('Confirm rejection of Concession Q-1042? This will freeze the quote.');
      });
    }

    if (returnBtn) {
      returnBtn.addEventListener('click', function() {
        if (!textarea.value.trim()) {
          textarea.focus();
          textarea.classList.add('border-error');
          alert('Please supply revision comments for sales rep M. Shah.');

      });

              ${text.replace(/</g, "&lt;").replace(/>/g, "&gt;")}

          concessionInput.value = '';

      if (downloadPdfBtn) {
        downloadPdfBtn.addEventListener('click', function () {
          showToast('Generating Document', 'Preparing formal PDF quotation package for Q-1042...');
        });
      }

        document.addEventListener('DOMContentLoaded', () => {
          // 1. Table Filter: SKU & Depot Logic
          const skuInput = document.getElementById('skuSearchInput');
          const depotSelect = document.getElementById('depotFilterSelect');
          const categorySelect = document.getElementById('categoryFilterSelect');
          const rows = document.querySelectorAll('.item-row');

          function filterLedger() {
            const query = skuInput.value.toLowerCase();
            const selectedDepot = depotSelect.value;
            const selectedCat = categorySelect.value;

            rows.forEach(row => {
              const rowDepot = row.getAttribute('data-depot');
              const rowSku = row.getAttribute('data-sku');
              const text = row.innerText.toLowerCase();

              const matchesSearch = !query || text.includes(query);
              const matchesDepot = selectedDepot === 'ALL' || rowDepot === selectedDepot;
              const matchesCat = selectedCat === 'ALL' || rowSku.startsWith(selectedCat);

              if (matchesSearch && matchesDepot && matchesCat) {
                row.style.display = '';
              } else {
                row.style.display = 'none';
              }
            });
          }

          skuInput.addEventListener('input', filterLedger);
          depotSelect.addEventListener('change', filterLedger);
          categorySelect.addEventListener('change', filterLedger);
